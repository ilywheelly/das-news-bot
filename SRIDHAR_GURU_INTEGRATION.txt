# Интеграция «Шлоки дня» в DAS.py

Минимальный пример:

from sridhar_guru import (
    extract_candidates_from_post,
    choose_unpublished,
    mark_published,
    telegram_html,
)

async def send_shloka_of_day(bot, lecture_url, channel_id):
    candidates = extract_candidates_from_post(lecture_url)
    shloka = choose_unpublished(candidates)

    if not shloka:
        print("На этой странице нет новой подходящей шлоки.")
        return

    text = telegram_html(shloka)

    await bot.send_message(
        chat_id=channel_id,
        text=text,
        parse_mode="HTML",
        disable_web_page_preview=True,
    )

    # Отмечаем опубликованной ТОЛЬКО после успешной отправки Telegram.
    mark_published(shloka)
