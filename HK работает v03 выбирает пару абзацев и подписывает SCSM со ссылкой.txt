# Получение смыслового ядра статьи
def get_full_article(link):
    headers = {"User-Agent": "bot_DAS/1.0"}
    try:
        response = requests.get(link, headers=headers)
        if response.status_code == 200:
            soup = BeautifulSoup(response.content, "html.parser")
            return extract_meaningful_paragraph(soup)
        else:
            logger.error(f"Ошибка при подключении к странице статьи, код ответа: {response.status_code}")
            return "Ошибка при получении статьи."
    except Exception as e:
        logger.error(f"Ошибка при обращении к странице статьи: {e}")
        return "Ошибка при получении статьи."

def is_russian(text):
    cyrillic = len(re.findall(r'[а-яА-Я]', text))
    total = len(text)
    return total > 0 and (cyrillic / total) > 0.7

def extract_meaningful_paragraph(soup):
    paragraphs = soup.find_all("p")
    keywords = ["гуру", "махапрабху", "преданность", "бхакти", "рамананда", "душа"]

    scored_paragraphs = []
    for i, para in enumerate(paragraphs):
        text = para.get_text(strip=True)
        if not text or len(text) < 80:
            continue
        if not is_russian(text):
            continue

        length_score = len(text)
        keyword_score = sum(text.lower().count(k) for k in keywords)
        position_score = -abs(i - len(paragraphs) // 2)
        total_score = length_score * 0.1 + keyword_score * 10 + position_score

        scored_paragraphs.append((total_score, text))

    scored_paragraphs.sort(reverse=True)
    best_paragraphs = [text for score, text in scored_paragraphs[:2]]
    return "\n\n".join(best_paragraphs) if best_paragraphs else "Описание не найдено."

# Обработка команды /hk
async def send_harekrishna_article_command(update, context):

    logger.info(f"Поиск статьи на тему: {query}")
    url = "https://harekrishna.ru/rss/read/"
    headers = {"User-Agent": "bot_DAS/1.0"}
    try:
        response = requests.get(url, headers=headers)
        if response.status_code == 200:
            soup = BeautifulSoup(response.content, "lxml-xml")
            items = soup.find_all("item")
            filtered_items = [item for item in items if query.lower() in item.find("title").text.lower()]
            if filtered_items:
                item = random.choice(filtered_items)
                title = item.find("title").text
                link = item.find("link").text
                description = get_full_article(link)

                message = f"{title}\n\n{description}\n\n[Подробнее]({link})\n\n{SOURCE_NAME}: {SOURCE_URL}"
                await context.bot.send_message(chat_id=CHANNEL_ID, text=message, parse_mode=ParseMode.MARKDOWN)
                logger.info(f"Статья по запросу '{query}' успешно отправлена в канал: {title}")
            else:
                await update.message.reply_text("Не удалось найти статью по вашему запросу.")
                logger.info(f"Статья по запросу '{query}' не найдена.")
        else:
            logger.error(f"Ошибка при подключении к сайту harekrishna.ru, код ответа: {response.status_code}")
    except Exception as e:
        logger.error(f"Ошибка при обращении к сайту harekrishna.ru: {e}")
